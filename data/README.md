```text
data/
├── gdsc/
│   ├── gdsc_expr.csv
│   ├── gdsc_sens.csv
│   ├── gdsc_smiles.csv
│   └── gdsc_targets.csv
├── gse112274/
│   ├── GSE112274_expr.csv
│   ├── GSE112274_label.csv
│   ├── target_smiles.csv
│   └── target_targets.csv
└── pathways/
    └── h.all.v2026.1.Hs.symbols.gmt
```
